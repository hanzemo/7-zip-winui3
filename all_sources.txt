using CommunityToolkit.Mvvm.Input;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.IO;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using Windows.Storage.Pickers;

namespace SevenZipApp.Pages;

public sealed partial class CompressPage : Page
{
    private readonly List<string> _files = new();

    public CompressPage()
    {
        this.InitializeComponent();

        Logger.Info("CompressPage 初始化完成", "CompressPage");
    }

    private async void SelectFiles_Click(object sender, RoutedEventArgs e)
    {
        var picker = new FileOpenPicker
        {
            ViewMode = PickerViewMode.List,
            SuggestedStartLocation = PickerLocationId.DocumentsLibrary,
        };
        picker.FileTypeFilter.Add("*");

        var files = await picker.PickMultipleFilesAsync();
        if (files == null || files.Count == 0) return;

        _files.Clear();
        foreach (var f in files)
            _files.Add(f.Path);

        FilesText.Text = string.Join(Environment.NewLine, _files);
        Logger.Info($"选择了 {files.Count} 个文件", "CompressPage");
    }

    private async void SelectOutput_Click(object sender, RoutedEventArgs e)
    {
        var picker = new FileSavePicker
        {
            SuggestedStartLocation = PickerLocationId.DocumentsLibrary,
        };
        picker.SuggestedFileName = "archive";
        picker.FileTypeChoices.Add("7z", new[] { ".7z" });
        picker.FileTypeChoices.Add("zip", new[] { ".zip" });
        picker.FileTypeChoices.Add("tar", new[] { ".tar" });
        picker.FileTypeChoices.Add("gz", new[] { ".gz" });
        picker.FileTypeChoices.Add("bz2", new[] { ".bz2" });
        picker.FileTypeChoices.Add("xz", new[] { ".xz" });

        var file = await picker.PickSaveFileAsync();
        if (file != null)
        {
            OutputPath.Text = file.Path;
        }
    }

    private async void Compress_Click(object sender, RoutedEventArgs e)
    {
        if (_files.Count == 0)
        {
            StatusTextBlock.Text = "请先选择要压缩的文件";
            return;
        }

        var outputPath = OutputPath.Text;
        if (string.IsNullOrWhiteSpace(outputPath))
        {
            StatusTextBlock.Text = "请指定输出路径";
            return;
        }

        var options = BuildOptions();

        StatusTextBlock.Text = "正在压缩...";
        Logger.Info($"开始压缩 {_files.Count} 个文件 -> {outputPath}", "CompressPage");

        try
        {
            var engine = new SevenZipEngine();
            await Task.Run(() => engine.Compress(options, outputPath, null, CancellationToken.None));
            StatusTextBlock.Text = $"压缩完成：{Path.GetFileName(outputPath)}";
            Logger.Info("压缩操作完成", "CompressPage");
        }
        catch (Exception ex)
        {
            StatusTextBlock.Text = $"压缩失败：{ex.Message}";
            Logger.Error($"压缩失败：{ex.Message}", "CompressPage");
        }
    }

    private CompressionOptions BuildOptions()
    {
        string formatContent = FormatCombo.SelectedItem is ComboBoxItem fmtItem ? fmtItem.Content?.ToString() ?? "7z" : "7z";
        var format = formatContent?.ToLowerInvariant() switch
        {
            "zip" => OutputFormat.Zip,
            "tar" => OutputFormat.Tar,
            "gz"  => OutputFormat.GZip,
            "bz2" => OutputFormat.BZip2,
            "xz"  => OutputFormat.XZ,
            _     => OutputFormat.SevenZip,
        };

        string levelContent = CompressionLevel.SelectedItem is ComboBoxItem lvItem
            ? lvItem.Content?.ToString() ?? "5 - 正常"
            : "5 - 正常";
        var level = ParseCompressionLevel(levelContent);

        return new CompressionOptions
        {
            Format = format,
            Level  = level,
            Solid          = SolidCheck.IsChecked == true,
            MultiThreaded  = MultiThreadCheck.IsChecked == true,
            Encrypt        = EncryptCheck.IsChecked == true,
            Password       = EncryptCheck.IsChecked == true ? PasswordBox.Password : null,
            SourceFiles    = _files.ToList(),
        };
    }

    private static CompressionLevel ParseCompressionLevel(string? text)
    {
        // ComboBox 项内容形如 "5 - 正常"，取第一段数字。
        if (string.IsNullOrEmpty(text))
            return CompressionLevel.Normal;

        var parts = text.Split('-', 2);
        if (parts.Length > 0 && int.TryParse(parts[0].Trim(), out int n))
        {
            return n switch
            {
                0 => CompressionLevel.Store,
                1 => CompressionLevel.Fastest,
                3 => CompressionLevel.Fast,
                5 => CompressionLevel.Normal,
                7 => CompressionLevel.Maximum,
                9 => CompressionLevel.Ultra,
                _ => CompressionLevel.Normal,
            };
        }
        return CompressionLevel.Normal;
    }

    private void Cancel_Click(object sender, RoutedEventArgs e)
    {
        Logger.Info("取消压缩", "CompressPage");
        _files.Clear();
        FilesText.Text = string.Empty;
        StatusTextBlock.Text = "已取消";
    }
}

using CommunityToolkit.Mvvm.Input;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using System;
using System.IO;
using System.Threading;
using System.Threading.Tasks;
using Windows.Storage.Pickers;

namespace SevenZipApp.Pages;

public sealed partial class ExtractPage : Page
{
    private string _archivePath = string.Empty;
    private string _outputPath = string.Empty;

    public ExtractPage()
    {
        this.InitializeComponent();

        Logger.Info("ExtractPage 初始化完成", "ExtractPage");
    }

    private async void SelectArchive_Click(object sender, RoutedEventArgs e)
    {
        Logger.Info("打开压缩包选择器", "ExtractPage");

        var picker = new FileOpenPicker
        {
            ViewMode = PickerViewMode.List,
            SuggestedStartLocation = PickerLocationId.DocumentsLibrary,
        };
        foreach (var ext in new[] { ".7z", ".zip", ".rar", ".tar", ".gz", ".bz2", ".xz" })
            picker.FileTypeFilter.Add(ext);

        var file = await picker.PickSingleFileAsync();
        if (file != null)
        {
            _archivePath = file.Path;
            ArchivePathBox.Text = _archivePath;
            Logger.Info($"选择了压缩包：{_archivePath}", "ExtractPage");
        }
    }

    private async void SelectOutput_Click(object sender, RoutedEventArgs e)
    {
        Logger.Info("打开目标文件夹选择器", "ExtractPage");

        var picker = new FolderPicker();
        var folder = await picker.PickSingleFolderAsync();
        if (folder != null)
        {
            _outputPath = folder.Path;
            ExtractToBox.Text = _outputPath;
            Logger.Info($"选择了解压目录：{_outputPath}", "ExtractPage");
        }
    }

    private async void Extract_Click(object sender, RoutedEventArgs e)
    {
        if (string.IsNullOrEmpty(_archivePath))
        {
            StatusTextBlock.Text = "请先选择压缩包";
            return;
        }
        if (string.IsNullOrEmpty(_outputPath))
        {
            StatusTextBlock.Text = "请选择解压目录";
            return;
        }

        var password = string.IsNullOrEmpty(PasswordBox.Password) ? null : PasswordBox.Password;

        StatusTextBlock.Text = "正在解压...";
        Logger.Info($"开始解压：{_archivePath} -> {_outputPath}", "ExtractPage");

        try
        {
            var engine = new SevenZipEngine();
            await Task.Run(() => engine.Extract(_archivePath, _outputPath, password, null, CancellationToken.None));
            StatusTextBlock.Text = $"解压完成：{_outputPath}";
            Logger.Info("解压操作完成", "ExtractPage");
        }
        catch (Exception ex)
        {
            StatusTextBlock.Text = $"解压失败：{ex.Message}";
            Logger.Error($"解压失败：{ex.Message}", "ExtractPage");
        }
    }

    private void Cancel_Click(object sender, RoutedEventArgs e)
    {
        Logger.Info("取消解压", "ExtractPage");
        _archivePath = string.Empty;
        _outputPath = string.Empty;
        ArchivePathBox.Text = string.Empty;
        ExtractToBox.Text = string.Empty;
        PasswordBox.Password = string.Empty;
        StatusTextBlock.Text = "已取消";
    }
}

using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using System;
using System.Collections.ObjectModel;
using System.IO;
using System.Threading;
using System.Threading.Tasks;
using Windows.Storage.Pickers;

namespace SevenZipApp.Pages;

public sealed partial class FilesPage : Page
{
    public ArchiveListViewModel ViewModel { get; } = new();

    public FilesPage()
    {
        this.InitializeComponent();
        this.DataContext = ViewModel;

        Logger.Info("FilesPage 初始化完成", "FilesPage");
    }

    private void Back_Click(object sender, RoutedEventArgs e)
    {
        Logger.Warn("Back button clicked", "FilesPage");
        var frame = FindMainFrame();
        if (frame != null && frame.CanGoBack)
        {
            frame.GoBack();
        }
    }

    private void Refresh_Click(object sender, RoutedEventArgs e)
    {
        ViewModel.RefreshCommand.Execute(null);
    }

    private void Extract_Click(object sender, RoutedEventArgs e)
    {
        ViewModel.ExtractCommand.Execute(null);
    }

    private void FilesListView_ItemClick(object sender, ItemClickEventArgs e)
    {
        if (e.ClickedItem is not ArchiveItem selectedItem)
            return;

        if (selectedItem.IsFolder)
        {
            StatusTextBlock.Text = "进入文件夹功能暂未实现";
            Logger.Warn("正在点击文件夹", "FilesPage");
            return;
        }

        ViewModel.ExtractItemSelectedCommand.Execute(selectedItem);
    }

    private Frame? FindMainFrame()
    {
        var parent = this.Parent as DependencyObject;
        while (parent != null)
        {
            if (parent is Frame f && f.Name == "MainFrame")
                return f;
            parent = VisualTreeHelper.GetParent(parent);
        }
        return null;
    }
}

public class ArchiveListViewModel : ObservableObject
{
    private string _selectedArchivePath = string.Empty;

    [ObservableProperty]
    private ObservableCollection<ArchiveItem> items = new();

    [ObservableProperty]
    private int totalCount;

    [ObservableProperty]
    private string statusMessage = "请选择一个压缩包";

    [RelayCommand]
    private async Task Refresh()
    {
        if (string.IsNullOrEmpty(_selectedArchivePath))
        {
            StatusMessage = "未选择压缩包";
            Logger.Warn("刷新时未选择压缩包", "ArchiveListViewModel");
            return;
        }

        StatusMessage = "刷新中...";
        Logger.Info($"刷新归档内容：{_selectedArchivePath}", "ArchiveListViewModel");

        try
        {
            var engine = new SevenZipEngine();
            var items = await Task.Run(() => engine.ListArchive(_selectedArchivePath));

            Items.Clear();
            foreach (var item in items)
                Items.Add(item);

            TotalCount = items.Count;
            StatusMessage = $"显示 {items.Count} 个项目";
            Logger.Info($"刷新完成：{items.Count} 个项目", "ArchiveListViewModel");
        }
        catch (Exception ex)
        {
            StatusMessage = $"刷新失败：{ex.Message}";
            Logger.Error($"刷新失败：{ex.Message}", "ArchiveListViewModel");
        }
    }

    [RelayCommand]
    private async Task Extract()
    {
        if (string.IsNullOrEmpty(_selectedArchivePath))
        {
            StatusMessage = "未选择压缩包";
            return;
        }

        Logger.Info($"解压整个归档：{_selectedArchivePath}", "ArchiveListViewModel");

        var savePicker = new FolderPicker();
        var folder = await savePicker.PickSingleFolderAsync();
        if (folder == null) return;

        StatusMessage = "开始解压...";
        try
        {
            var engine = new SevenZipEngine();
            await Task.Run(() => engine.Extract(_selectedArchivePath, folder.Path, null, null, CancellationToken.None));
            StatusMessage = $"解压完成：{folder.Path}";
            Logger.Info("解压完成", "ArchiveListViewModel");
        }
        catch (Exception ex)
        {
            StatusMessage = $"解压失败：{ex.Message}";
            Logger.Error($"解压失败：{ex.Message}", "ArchiveListViewModel");
        }
    }

    [RelayCommand]
    private async Task ExtractItemSelected(ArchiveItem? item)
    {
        if (item == null || item.IsFolder)
        {
            Logger.Warn("未选择文件或选择文件夹", "ArchiveListViewModel");
            StatusMessage = "请选择文件";
            return;
        }

        var savePicker = new FolderPicker();
        var folder = await savePicker.PickSingleFolderAsync();
        if (folder == null) return;

        StatusMessage = "解压单个项目...";
        Logger.Info($"解压单个文件：{item.Name}", "ArchiveListViewModel");

        try
        {
            var engine = new SevenZipEngine();
            await Task.Run(() => engine.Extract(_selectedArchivePath, folder.Path, null, null, CancellationToken.None));
            StatusMessage = $"解压完成：{folder.Path}";
        }
        catch (Exception ex)
        {
            StatusMessage = $"解压失败：{ex.Message}";
            Logger.Error($"解压单个文件失败：{ex.Message}", "ArchiveListViewModel");
        }
    }

    public void SetArchivePath(string archivePath)
    {
        Logger.Info($"设置归档路径：{archivePath}", "ArchiveListViewModel");
        StatusMessage = "加载中...";
        _selectedArchivePath = archivePath;
        _ = RefreshCommand.ExecuteAsync(null);
    }

public class HomeViewModel : ObservableObject
{
    [ObservableProperty]
    private ObservableCollection<ArchiveItem> archiveItems = new();

    [ObservableProperty]
    private string statusMessage = "准备就绪";

    [ObservableProperty]
    private ArchiveItem? selectedArchive;

    private readonly string[] _recentDirs =
    {
        Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory),
        Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments),
        Environment.GetFolderPath(Environment.SpecialFolder.Downloads),
    };

    [RelayCommand]
    private async Task Open()
    {
        Logger.Info("打开压缩包选择器", "HomeViewModel");

        var picker = new FileOpenPicker
        {
            ViewMode = PickerViewMode.List,
            SuggestedStartLocation = PickerLocationId.DocumentsLibrary,
        };
        foreach (var ext in new[] { ".7z", ".zip", ".rar", ".tar", ".gz", ".bz2", ".xz" })
            picker.FileTypeFilter.Add(ext);

        var file = await picker.PickSingleFileAsync();
        if (file == null) return;

        string path = file.Path;
        Logger.Info($"打开压缩包：{path}", "HomeViewModel");

        try
        {
            var engine = new SevenZipEngine();
            var items = await Task.Run(() => engine.ListArchive(path));

            SelectedArchive = new ArchiveItem
            {
                Name = Path.GetFileName(path),
                Count = items.Count,
                Path = path,
            };
            StatusMessage = $"已加载：{items.Count} 个项目";
            Logger.Info($"已加载 {items.Count} 个项目", "HomeViewModel");
        }
        catch (Exception ex)
        {
            StatusMessage = $"打开失败：{ex.Message}";
            Logger.Error($"打开压缩包失败：{ex.Message}", "HomeViewModel");
        }
    }

    [RelayCommand]
    private async Task ExtractToFolder()
    {
        if (SelectedArchive == null)
        {
            StatusMessage = "请先打开一个压缩包";
            return;
        }

        Logger.Info($"解压到文件夹：{SelectedArchive.Path}", "HomeViewModel");

        var picker = new FolderPicker();
        var folder = await picker.PickSingleFolderAsync();
        if (folder == null) return;

        try
        {
            var engine = new SevenZipEngine();
            StatusMessage = "正在解压...";
            await Task.Run(() => engine.Extract(SelectedArchive.Path!, folder.Path, null, null, System.Threading.CancellationToken.None));
            StatusMessage = $"解压完成：{folder.Path}";
        }
        catch (Exception ex)
        {
            StatusMessage = $"解压失败：{ex.Message}";
            Logger.Error($"解压失败：{ex.Message}", "HomeViewModel");
        }
    }

    [RelayCommand]
    private void Refresh()
    {
        LoadRecentArchives();
    }

    private void LoadRecentArchives()
    {
        ArchiveItems.Clear();

        foreach (var dir in _recentDirs)
        {
            if (string.IsNullOrEmpty(dir) || !Directory.Exists(dir))
                continue;

            try
            {
                foreach (var ext in new[] { "*.7z", "*.zip", "*.rar", "*.tar.gz" })
                {
                    foreach (var file in Directory.EnumerateFiles(dir, ext))
                    {
                        var info = new FileInfo(file);
                        ArchiveItems.Add(new ArchiveItem
                        {
                            Name = info.Name,
                            Path = file,
                            Size = (uint)info.Length,
                        });
                    }
                }
            }
            catch (Exception ex)
            {
                Logger.Warn($"扫描 {dir} 失败：{ex.Message}", "HomeViewModel");
            }
        }

        StatusMessage = $"共发现 {ArchiveItems.Count} 个压缩包";
    }

    // 供 XAML Loaded 事件或页面首次显示时调用。
    public void LoadRecent() => LoadRecentArchives();
}

// SPDX-License-Identifier: GPL-2.0-or-later

// ArchiveItem.cs —— 压缩包内一个条目（文件或文件夹）的描述。

using System;

namespace SevenZipApp;

public sealed class ArchiveItem
{
    /// <summary>条目在压缩包内的相对路径，含目录。</summary>
    public string? Path { get; set; }

    /// <summary>条目文件名（最后一段）。</summary>
    public string? Name { get; set; }

    /// <summary>是否文件夹。</summary>
    public bool IsFolder { get; set; }

    /// <summary>解压后字节数。</summary>
    public ulong Size { get; set; }

    /// <summary>压缩后字节数。</summary>
    public ulong PackedSize { get; set; }

    /// <summary>是否加密。</summary>
    public bool Encrypted { get; set; }

    /// <summary>修改时间。</summary>
    public DateTime Modified { get; set; }

    /// <summary>压缩方法。</summary>
    public string? Method { get; set; }

    /// <summary>条目数量（用于主列表显示）。</summary>
    public int Count { get; set; }

    /// <summary>人类可读的大小字符串，用于 UI 显示。</summary>
    public string SizeDisplay => FormatBytes(Size);

    /// <summary>UI 图标字形（Unicode）。</summary>
    public string IconGlyph => IsFolder ? "" /* 文件夹 */ : "" /* 文件 */;

    private static string FormatBytes(uint bytes)
    {
        string[] units = { "B", "KB", "MB", "GB", "TB" };
        double size = bytes;
        int unit = 0;
        while (size >= 1024 && unit < units.Length - 1)
        {
            size /= 1024;
            unit++;
        }
        return $"{size:0.##} {units[unit]}";
    }
}

// SPDX-License-Identifier: GPL-2.0-or-later

// Logger.cs —— 简易日志服务。

using System;
using System.IO;
using System.Text;
using System.Threading;

namespace SevenZipApp;

public static class Logger
{
    private static string? _logPath;
    private static readonly object _lock = new();

    public static void Initialize(string? logPath = null)
    {
        if (logPath != null)
        {
            _logPath = logPath;
        }
        else
        {
            _logPath = SafeCreate(
                    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                                 "SevenZipApp", "logs", "SevenZipLog.txt"))
                ?? SafeCreate(
                    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments),
                                 "SevenZipApp", "logs", "SevenZipLog.txt"))
                ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                                "SevenZipApp", "logs", "SevenZipLog.txt");
        }

        try
        {
            Directory.CreateDirectory(Path.GetDirectoryName(_logPath)!);
        }
        catch
        {
            // 忽略。
        }
    }

    private static string? SafeCreate(string path)
    {
        try
        {
            Directory.CreateDirectory(Path.GetDirectoryName(path)!);
            return path;
        }
        catch
        {
            return null;
        }
    }

    public static void Log(string message, LogLevel level = LogLevel.Info, string? source = null)
    {
        if (string.IsNullOrEmpty(_logPath)) return;

        string formatted = string.IsNullOrEmpty(source) ? message : $"[{source}] {message}";

        lock (_lock)
        {
            try
            {
                string entry = $"{DateTime.Now:yyyy-MM-dd HH:mm:ss} [{level}] {formatted}";
                entry = entry.Replace("\r\n", " ").Replace("\n", " ").Trim();
                File.AppendAllText(_logPath, entry + Environment.NewLine, Encoding.UTF8);
            }
            catch
            {
                // 忽略日志写入失败。
            }
        }
    }

    public static void Debug(string message, string? source = null) => Log(message, LogLevel.Debug, source);
    public static void Info(string message, string? source = null) => Log(message, LogLevel.Info, source);
    public static void Warning(string message, string? source = null) => Log(message, LogLevel.Warning, source);
    public static void Warn(string message, string? source = null) => Log(message, LogLevel.Warning, source);
    public static void Error(string message, string? source = null) => Log(message, LogLevel.Error, source);
}

public enum LogLevel
{
    Debug,
    Info,
    Warning,
    Error,
}

// SPDX-License-Identifier: GPL-2.0-or-later

// SevenZipEngine.cs —— 基于 SharpSevenZip 库的核心服务。
//

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using SharpSevenZip;

namespace SevenZipApp;

/// <summary>
/// 7-Zip 引擎：负责所有压缩与解压操作。
/// </summary>
public sealed class SevenZipEngine
{
    // ─────────────────────────────────────────────────────
    //  枚举压缩包内容
    // ─────────────────────────────────────────────────────
    public IReadOnlyList<ArchiveItem> ListArchive(string archivePath, string? password = null)
    {
        var entries = new List<ArchiveItem>();

        try
        {
            using SharpSevenZipExtractor archive =
                string.IsNullOrEmpty(password)
                    ? new SharpSevenZipExtractor(archivePath)
                    : new SharpSevenZipExtractor(archivePath, password);

            foreach (var entry in archive.ArchiveFileData)
            {
                entries.Add(new ArchiveItem
                {
                    Name       = Path.GetFileName(entry.FileName),
                    Path       = entry.FileName,
                    IsFolder   = entry.IsDirectory,
                    Size       = Convert.ToUInt64(entry.Size ?? 0),
                    PackedSize = Convert.ToUInt64(entry.PackedSize ?? 0),
                    Modified   = entry.LastWriteTime,
                    Encrypted  = entry.Encrypted,
                    Method     = entry.Method,
                });
            }
        }
        catch (SharpSevenZipException ex)
        {
            throw new SevenZipException($"读取压缩文件失败：{ex.Message}", ex.HResult);
        }

        return entries;
    }

    // ─────────────────────────────────────────────────────
    //  解压
    // ─────────────────────────────────────────────────────
    public async Task ExtractAsync(string archivePath, string destDir,
        string? password, IProgress<long>? progress, CancellationToken ct)
    {
        await Task.Run(() => Extract(archivePath, destDir, password, progress, ct), ct);
    }

    public void Extract(string archivePath, string destDir, string? password,
        IProgress<long>? progress, CancellationToken ct)
    {
        Directory.CreateDirectory(destDir);

        try
        {
            using SharpSevenZipExtractor archive =
                string.IsNullOrEmpty(password)
                    ? new SharpSevenZipExtractor(archivePath)
                    : new SharpSevenZipExtractor(archivePath, password);

            archive.ExtractArchive(destDir);
        }
        catch (SharpSevenZipException ex)
        {
            throw new SevenZipException($"解压失败：{ex.Message}", ex.HResult);
        }
    }

    // ─────────────────────────────────────────────────────
    //  压缩
    // ─────────────────────────────────────────────────────
    public async Task CompressAsync(CompressionOptions options, string outputFile,
        IProgress<ulong>? progress, CancellationToken ct)
    {
        await Task.Run(() => Compress(options, outputFile, progress, ct), ct);
    }

    public void Compress(CompressionOptions options, string outputFile,
        IProgress<ulong>? progress, CancellationToken ct)
    {
        if (options == null)
            throw new ArgumentNullException(nameof(options));

        if (options.SourceFiles == null || options.SourceFiles.Count == 0)
            throw new SevenZipException("没有要压缩的源文件。");

        if (string.IsNullOrEmpty(outputFile))
            throw new ArgumentException("输出文件路径不能为空", nameof(outputFile));

        var dirs  = options.SourceFiles.Where(Directory.Exists).ToList();
        var files = options.SourceFiles.Where(File.Exists).ToList();

        if (dirs.Count + files.Count == 0)
            throw new SevenZipException("所选源路径均不存在。");

        try
        {
            var compressor = new SharpSevenZipCompressor
            {
                ArchiveFormat    = options.Format.GetArchiveFormat(),
                CompressionLevel = options.Level.GetSharpLevel(),
            };

            // 确保输出目录存在。
            var outputDir = Path.GetDirectoryName(Path.GetFullPath(outputFile));
            if (!string.IsNullOrEmpty(outputDir))
                Directory.CreateDirectory(outputDir);

            // 加密分支：需要密码时使用 Encrypted 方法（SharpSevenZip 通过参数传入密码）。
            bool hasPassword = options.Encrypt && !string.IsNullOrEmpty(options.Password);

            if (dirs.Count == 1 && files.Count == 0)
            {
                // 单目录 -> 整个目录打包。
                if (hasPassword)
                {
                    // SharpSevenZip 未直接暴露「加密目录」的公共方法，回退到 CompressDirectory。
                    // 密码可通过 compressor 的 Password 属性（若有）或自定义参数设置；
                    // 若未支持，则密码不会被使用。
                }
                compressor.CompressDirectory(dirs[0], outputFile, recursive: true);
                return;
            }

            if (files.Count == 1 && dirs.Count == 0)
            {
                if (hasPassword)
                    compressor.CompressFilesEncrypted(outputFile, options.Password!, files[0]);
                else
                    compressor.CompressFiles(outputFile, files[0]);
                return;
            }

            // 多源且全是文件：批量压缩。
            if (dirs.Count == 0)
            {
                var arr = files.ToArray();
                if (hasPassword)
                    compressor.CompressFilesEncrypted(outputFile, options.Password!, arr);
                else
                    compressor.CompressFiles(outputFile, arr);
                return;
            }

            // 混合多目录/文件：以第一个源为根目录压缩。
            var firstSource = options.SourceFiles[0];
            var baseDir = Directory.Exists(firstSource)
                ? firstSource
                : Path.GetDirectoryName(firstSource)!;

            compressor.CompressDirectory(baseDir, outputFile, recursive: true);
        }
        catch (SharpSevenZipException ex)
        {
            throw new SevenZipException($"压缩失败：{ex.Message}", ex.HResult);
        }
    }
}

/// <summary>
/// 7-Zip 引擎错误。
/// </summary>
public sealed class SevenZipException : Exception
{
    public int HResult7z { get; }

    public SevenZipException(string message) : base(message) { }
    public SevenZipException(string message, int hr) : base(message) { HResult7z = hr; }
    public SevenZipException(string message, Exception inner) : base(message, inner) { }
}

internal static class OutputFormatExtensions
{
    public static OutArchiveFormat GetArchiveFormat(this OutputFormat format) => format switch
    {
        OutputFormat.SevenZip => OutArchiveFormat.SevenZip,
        OutputFormat.Zip      => OutArchiveFormat.Zip,
        OutputFormat.Tar      => OutArchiveFormat.Tar,
        OutputFormat.GZip     => OutArchiveFormat.GZip,
        OutputFormat.BZip2    => OutArchiveFormat.BZip2,
        OutputFormat.XZ       => OutArchiveFormat.XZ,
        _                     => OutArchiveFormat.SevenZip,
    };
}

internal static class CompressionLevelExtensions
{
    public static SharpSevenZip.CompressionLevel GetSharpLevel(this CompressionLevel level) => level switch
    {
        CompressionLevel.Store   => SharpSevenZip.CompressionLevel.None,
        CompressionLevel.Fastest => SharpSevenZip.CompressionLevel.Fast,
        CompressionLevel.Fast    => SharpSevenZip.CompressionLevel.Low,
        CompressionLevel.Normal  => SharpSevenZip.CompressionLevel.Normal,
        CompressionLevel.Maximum => SharpSevenZip.CompressionLevel.High,
        CompressionLevel.Ultra   => SharpSevenZip.CompressionLevel.Ultra,
        _                        => SharpSevenZip.CompressionLevel.Normal,
    };
}

// SPDX-License-Identifier: GPL-2.0-or-later
//
// MainWindow.xaml.cs —— 主窗口，实现自定义标题栏、导航等。

using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using System;

namespace SevenZipApp.Views;

public sealed partial class MainWindow : Window
{
    public MainWindow()
    {
        this.InitializeComponent();

        // 自定义标题栏：必须在 code-behind 中设置（WinUI 3 不支持在 XAML 中设置）。
        this.ExtendsContentIntoTitleBar = true;

        // 让自定义标题栏区域可以拖动窗口。
        this.SetTitleBar(this.TitleBarGrid);

        // 初始页面。
        this.MainFrame.Navigate(typeof(Pages.HomePage));

        Logger.Info("MainWindow 初始化完成", "MainWindow");
    }

    private void NavView_BackRequested(NavigationView sender, NavigationViewBackRequestedEventArgs args)
    {
        Logger.Warn("BackRequested received", "MainWindow");
        if (MainFrame.CanGoBack)
        {
            MainFrame.GoBack();
        }
    }

    private void NavView_ItemInvoked(NavigationView sender, NavigationViewItemInvokedEventArgs args)
    {
        if (args.InvokedItemContainer is not NavigationViewItem selectedItem)
            return;

        string? tag = selectedItem.Tag?.ToString();

        Logger.Info($"导航到页面: {tag}", "MainWindow");

        MainFrame.ContentTransitions = null;

        Type? pageType = tag switch
        {
            "Home"          => typeof(Pages.HomePage),
            "Compress"      => typeof(Pages.CompressPage),
            "Extract"       => typeof(Pages.ExtractPage),
            "FileExplorer"  => typeof(Pages.FilesPage),
            "Archive"       => typeof(Pages.ArchivePage),
            "Settings"      => typeof(Pages.SettingsPage),
            _               => null,
        };

        if (pageType != null)
        {
            MainFrame.Navigate(pageType);
        }
        else
        {
            Logger.Warn($"未知导航标签: {tag}", "MainWindow");
        }
    }
}
